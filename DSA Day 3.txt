#include <stdio.h>
#define SIZE 5

typedef struct {
    int items[SIZE];
    int front;
    int rear;
} CircularQueue;

// Function to check if the queue is full
int isFull(CircularQueue* q) {
    return (q->front == (q->rear + 1) % SIZE);
}

// Function to check if the queue is empty
int isEmpty(CircularQueue* q) {
    return (q->front == -1);
}

// Insertion logic
void enQueue(CircularQueue* q, int value) {
    if (isFull(q)) {
        printf("Queue is full!\n");
        return;
    }
    
    if (q->front == -1) {
        q->front = 0; // Queue is initially empty, set front to 0
    }

    q->rear = (q->rear + 1) % SIZE; // Circular increment of rear
    q->items[q->rear] = value;
    printf("Inserted %d\n", value);
}

// Deletion logic
int deQueue(CircularQueue* q) {
    int element;
    
    if (isEmpty(q)) {
        printf("Queue is empty!\n");
        return -1;
    }

    element = q->items[q->front];
    
    if (q->front == q->rear) {
        // Queue is empty after deletion
        q->front = -1;
        q->rear = -1;
    } else {
        // Circular increment of front
        q->front = (q->front + 1) % SIZE;
    }

    printf("Deleted %d\n", element);
    return element;
}

// Display logic
void display(CircularQueue* q) {
    int i;
    
    if (isEmpty(q)) {
        printf("Queue is empty!\n");
        return;
    }
    
    printf("Front -> %d\n", q->front);
    printf("Items -> ");
    
    for (i = q->front; i != q->rear; i = (i + 1) % SIZE) {
        printf("%d ", q->items[i]);
    }
    printf("%d\n", q->items[i]); // Print the last element at the rear
    printf("Rear -> %d\n", q->rear);
}

int main() {
    CircularQueue q = {{0}, -1, -1}; // Initialize queue
    
    enQueue(&q, 1);
    enQueue(&q, 2);
    enQueue(&q, 3);
    enQueue(&q, 4);
    enQueue(&q, 5);
    
    display(&q);
    
    deQueue(&q);
    display(&q);
    
    enQueue(&q, 6);
    display(&q);
    
    return 0;
}
